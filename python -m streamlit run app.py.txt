# --- FINANCIAL CALCULATIONS (Using Session State) ---
sq_ft = st.session_state.sq_ft
current_cost_sf = st.session_state.cost_per_sqft
total_build_cost = sq_ft * current_cost_sf
total_cost = total_build_cost + st.session_state.const_closing_costs
equity_required = total_cost * 0.30
loan_amount = total_cost - equity_required

refi_points_cost = loan_amount * (st.session_state.refi_points / 100.0)
total_cash_invested = equity_required + st.session_state.const_closing_costs + refi_points_cost + st.session_state.refi_closing_costs

# Income, Vacancy, EGI, OpEx & NOI Calculations
gross_potential_income = st.session_state.market_rent * 12.0
vacancy_loss_dollars = gross_potential_income * (st.session_state.vacancy_rate / 100.0)
effective_gross_income = gross_potential_income - vacancy_loss_dollars

total_annual_opex = st.session_state.opex_df["Annual Cost ($)"].sum() if "Annual Cost ($)" in st.session_state.opex_df.columns else 0.0
opex_ratio = (total_annual_opex / effective_gross_income * 100.0) if effective_gross_income > 0 else 0.0

net_operating_income = effective_gross_income - total_annual_opex

# Debt Service & Returns
monthly_interest_rate = (st.session_state.refi_rate / 100.0) / 12.0
monthly_debt_service = (loan_amount * (monthly_interest_rate * (1 + monthly_interest_rate) ** 360) / 
                       ((1 + monthly_interest_rate) ** 360 - 1)) if monthly_interest_rate > 0 else (loan_amount / 360)
annual_debt_service = monthly_debt_service * 12.0

dscr = net_operating_income / annual_debt_service if annual_debt_service > 0 else 0.0
annual_cash_flow = net_operating_income - annual_debt_service
cash_on_cash = (annual_cash_flow / total_cash_invested * 100.0) if total_cash_invested > 0 else 0.0

# Breakeven Occupancy Calculation
breakeven_occupancy = ((total_annual_opex + annual_debt_service) / gross_potential_income * 100.0) if gross_potential_income > 0 else 0.0
