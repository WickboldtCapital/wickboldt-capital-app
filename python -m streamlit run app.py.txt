# Page Config
st.set_page_config(
    page_title="Proforma Development",
    layout="wide",
    initial_sidebar_state="expanded",
)

# --- TOP HEADER: LOGO AND PORTAL TITLE ---
col_logo, col_title = st.columns([1, 6])
with col_logo:
  try:
    st.image("logo.png", width=150) # Ensure your logo file is named logo.png
  except:
    st.write("Logo Missing")

with col_title:
  st.title("Proforma Development")

# --- TOP CORNER LOGOUT / USER BAR ---
if st.session_state.get("logged_in", False):
  st.sidebar.markdown(f"**Current User:** {st.session_state['email']}")
  if st.sidebar.button("🚪 Sign Out", key="top_sign_out_btn"):
    st.session_state["logged_in"] = False
    st.session_state["email"] = ""
    st.session_state["role"] = "User"
    st.rerun()

st.markdown("---")
